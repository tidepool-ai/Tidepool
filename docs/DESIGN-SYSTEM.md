# Tidepool Design System

Design system for **Tidepool**, the block-based workspace (Notion-style) built with **Angular** (frontend) and **Python / FastAPI** (backend), with AI powered by a hosted **GLM** model over an OpenAI-compatible API.

- Default scheme: **Tidepool** (teal + orange, indigo for AI)
- 5 color schemes × 2 modes (light / dark), plus *System* mode and a *High contrast* option
- All schemes meet **WCAG 2.2 AA** text contrast (4.5:1) in both modes — verified

---

## 1. Principles

1. **Content first, quiet chrome.** Handles, menus and "+" buttons appear on hover or focus.
2. **Keyboard first.** Every action is reachable from the command bar (`Ctrl/Cmd + K`), the slash menu, or a shortcut.
3. **One primitive.** Blocks look and behave the same in pages, database rows and templates.
4. **AI is always visible as AI.** Anything generated by GLM uses the AI color, is labelled, cites its sources, and needs a human Accept before it becomes content.
5. **Flat, bordered surfaces.** Borders, not shadows, separate surfaces. Shadows only for things that float (dragged cards, popovers).

### 5.6 High contrast option

Applies on top of any scheme and mode: `--color-border` and `--color-muted` take the value of `--color-text`, focus rings become 3px.

### 5.7 Contrast check

Checked for every scheme in both modes, all passing:

| Pair | Minimum |
| --- | --- |
| text on bg, text on surface | 4.5:1 |
| muted on bg, muted on surface | 4.5:1 |
| on-primary on primary, on-accent on accent | 4.5:1 |
| ai-text on ai-soft, soft-text on soft | 4.5:1 |
| ai (icons, non-text) on surface | 3:1 |

Add a contrast unit test in CI so a new scheme cannot ship below these ratios.

## 6. Color scheme changer (Settings › Appearance)

### 6.1 User-facing behavior

| Control | Options | Default |
| --- | --- | --- |
| Mode | Light, Dark, System | System |
| Color scheme | Tidepool, Graphite, Orchard, Dusk, Paper | Tidepool |
| Text size | Small, Default, Large | Default |
| High contrast | On / Off | Off |
| Use for the whole workspace (admins) | On / Off | Off |

- Changes apply instantly with a live preview; no Save button.
- *System* follows `prefers-color-scheme` and updates live when the OS switches.
- Preference syncs to every device for the signed-in user.
- Precedence: **user choice › workspace default › Tidepool + System**.
- Each scheme card shows a mini thumbnail (rail, text, primary, accent, AI) rendered in the current mode.

### 6.3 Angular `ThemeService` (signals)

```ts
// core/theme/theme.service.ts
import { Injectable, effect, inject, signal, computed, DestroyRef } from '@angular/core';
import { DOCUMENT } from '@angular/common';
import { HttpClient } from '@angular/common/http';

export type Scheme = 'tidepool' | 'graphite' | 'orchard' | 'dusk' | 'paper';
export type Mode = 'light' | 'dark' | 'system';
export type TextSize = 'small' | 'default' | 'large';

export interface Appearance {
  scheme: Scheme;
  mode: Mode;
  textSize: TextSize;
  highContrast: boolean;
}

const DEFAULTS: Appearance = { scheme: 'tidepool', mode: 'system', textSize: 'default', highContrast: false };
const STORAGE_KEY = 'tidepool.appearance';
const FONT_BASE: Record<TextSize, string> = { small: '13px', default: '14px', large: '16px' };

@Injectable({ providedIn: 'root' })
export class ThemeService {
  private doc = inject(DOCUMENT);
  private http = inject(HttpClient);
  private media = this.doc.defaultView!.matchMedia('(prefers-color-scheme: dark)');
  private osDark = signal(this.media.matches);

  readonly appearance = signal<Appearance>(this.readLocal());
  readonly resolvedMode = computed(() =>
    this.appearance().mode === 'system' ? (this.osDark() ? 'dark' : 'light') : this.appearance().mode,
  );

  constructor() {
    const onChange = (e: MediaQueryListEvent) => this.osDark.set(e.matches);
    this.media.addEventListener('change', onChange);
    inject(DestroyRef).onDestroy(() => this.media.removeEventListener('change', onChange));

    // Apply to <html> whenever anything changes
    effect(() => {
      const a = this.appearance();
      const root = this.doc.documentElement;
      root.dataset['scheme'] = a.scheme;
      root.dataset['mode'] = this.resolvedMode();
      root.dataset['contrast'] = a.highContrast ? 'high' : 'normal';
      root.style.setProperty('--font-base', FONT_BASE[a.textSize]);
      try { localStorage.setItem(STORAGE_KEY, JSON.stringify(a)); } catch { /* private mode */ }
    });
  }

  /** Load server preference after login (server wins over local cache). */
  load() {
    this.http.get<Appearance>('/api/me/preferences/appearance')
      .subscribe(a => this.appearance.set({ ...DEFAULTS, ...a }));
  }

  update(patch: Partial<Appearance>) {
    this.doc.documentElement.classList.add('theme-transition');
    this.appearance.update(a => ({ ...a, ...patch }));
    this.http.patch('/api/me/preferences/appearance', patch).subscribe();
  }

  private readLocal(): Appearance {
    try { return { ...DEFAULTS, ...JSON.parse(localStorage.getItem(STORAGE_KEY) ?? '{}') }; }
    catch { return DEFAULTS; }
  }
}
```

Prevent a flash of the wrong theme by setting the attributes before Angular boots:

```html
<!-- index.html, inside <head> -->
<script>
  (function () {
    try {
      var a = JSON.parse(localStorage.getItem('tidepool.appearance') || '{}');
      var dark = a.mode === 'dark' || (a.mode !== 'light' && matchMedia('(prefers-color-scheme: dark)').matches);
      var r = document.documentElement;
      r.dataset.scheme = a.scheme || 'tidepool';
      r.dataset.mode = dark ? 'dark' : 'light';
      r.dataset.contrast = a.highContrast ? 'high' : 'normal';
    } catch (e) {}
  })();
</script>
```

### 6.4 Backend (FastAPI)

```python
# app/preferences/router.py
from typing import Literal
from fastapi import APIRouter, Depends
from pydantic import BaseModel
from sqlalchemy.ext.asyncio import AsyncSession

from app.auth import current_user
from app.db import get_session
from app.models import User, Workspace

router = APIRouter(prefix="/api/me/preferences", tags=["preferences"])

Scheme = Literal["tidepool", "graphite", "orchard", "dusk", "paper"]
Mode = Literal["light", "dark", "system"]
TextSize = Literal["small", "default", "large"]


class Appearance(BaseModel):
    scheme: Scheme = "tidepool"
    mode: Mode = "system"
    text_size: TextSize = "default"
    high_contrast: bool = False


class AppearancePatch(BaseModel):
    scheme: Scheme | None = None
    mode: Mode | None = None
    text_size: TextSize | None = None
    high_contrast: bool | None = None


@router.get("/appearance", response_model=Appearance)
async def get_appearance(user: User = Depends(current_user), db: AsyncSession = Depends(get_session)):
    workspace: Workspace = await db.get(Workspace, user.current_workspace_id)
    # Precedence: user › workspace default › built-in default
    return Appearance(**{**(workspace.default_appearance or {}), **(user.preferences.get("appearance") or {})})


@router.patch("/appearance", response_model=Appearance)
async def patch_appearance(
    patch: AppearancePatch,
    user: User = Depends(current_user),
    db: AsyncSession = Depends(get_session),
):
    current = user.preferences.get("appearance") or {}
    user.preferences = {**user.preferences, "appearance": {**current, **patch.model_dump(exclude_none=True)}}
    await db.commit()
    return Appearance(**user.preferences["appearance"])
```

Storage: `users.preferences JSONB` and `workspaces.default_appearance JSONB` (admin-only write via `PUT /api/workspaces/{id}/appearance`). Angular uses camelCase; configure Pydantic with an alias generator so the API returns `textSize` / `highContrast`.

## 8. Motion

- 150ms for hover/press, 200ms for theme change, 200–250ms for popovers (exit faster than enter).
- Streaming AI text: caret blink and 3-dot pulse only.
- Respect `prefers-reduced-motion`: disable all transitions and animations.
